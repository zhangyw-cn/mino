# Mino Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Go CLI (`mino`) that serves a local browser UI to browse, filter, and iframe-preview single-file HTML apps under a workspace directory, with live updates via filesystem watching and SSE.

**Architecture:** Single Go binary: CLI loads `.mino/config.toml`, scans HTML into an in-memory catalog, starts an HTTP server on `127.0.0.1`, embeds the UI, serves catalogued files under `/apps/*`, and pushes catalog changes over SSE. A filesystem watcher incrementally updates the catalog.

**Tech Stack:** Go 1.22+, `github.com/BurntSushi/toml`, `github.com/fsnotify/fsnotify`, `github.com/bmatcuk/doublestar/v4` (ignore globs), stdlib `net/http` + `embed`.

## Global Constraints

- Bind default host `127.0.0.1`; default port `0` (ephemeral).
- Only `.html` / `.htm` files are app entries; no companion static asset serving.
- Built-in ignores (fixed): `.mino/`, `.git/`, `.hg/`, `.svn/`, `node_modules/`.
- Config lives at `<root>/.mino/config.toml`; corrupt config fails loudly (no silent overwrite).
- CLI `--port` overrides config for the run and does not write back.
- Relative path (slash-separated, no leading `/`) is the unique id for an entry.
- Serve `/apps/*` only for paths present in the catalog; reject traversal.
- Read-only tool: no create/rename/delete/edit APIs.
- UI: tree + path/filename search + iframe preview; auto-refresh list and preview on disk changes.
- Module path: `mino` (local module).
- Tests: `go test ./...`; TDD per task.

```go
package config

import (
	"fmt"
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"
)

type Config struct {
	Name   string   `toml:"name"`
	Port   int      `toml:"port"`
	Host   string   `toml:"host"`
	Ignore []string `toml:"ignore"`
}

func defaultConfig(root string) Config {
	return Config{
		Name:   filepath.Base(root),
		Port:   0,
		Host:   "127.0.0.1",
		Ignore: nil,
	}
}

func Load(root string) (Config, error) {
	dir := filepath.Join(root, ".mino")
	path := filepath.Join(dir, "config.toml")
	if _, err := os.Stat(path); os.IsNotExist(err) {
		cfg := defaultConfig(root)
		if err := os.MkdirAll(dir, 0o755); err != nil {
			return Config{}, err
		}
		f, err := os.Create(path)
		if err != nil {
			return Config{}, err
		}
		defer f.Close()
		if err := toml.NewEncoder(f).Encode(cfg); err != nil {
			return Config{}, err
		}
		return cfg, nil
	} else if err != nil {
		return Config{}, err
	}

	var cfg Config
	meta, err := toml.DecodeFile(path, &cfg)
	if err != nil {
		return Config{}, fmt.Errorf("invalid config %s: %w", path, err)
	}
	_ = meta
	if cfg.Host == "" {
		cfg.Host = "127.0.0.1"
	}
	if cfg.Name == "" {
		cfg.Name = filepath.Base(root)
	}
	return cfg, nil
}
```

```go
package ignore

import (
	"path"
	"strings"

	"github.com/bmatcuk/doublestar/v4"
)

var builtin = []string{
	".mino",
	".mino/**",
	".git",
	".git/**",
	".hg",
	".hg/**",
	".svn",
	".svn/**",
	"node_modules",
	"node_modules/**",
	"**/.git/**",
	"**/.hg/**",
	"**/.svn/**",
	"**/node_modules/**",
	"**/.mino/**",
}

type Matcher struct {
	patterns []string
}

func New(extra []string) (*Matcher, error) {
	patterns := append([]string{}, builtin...)
	patterns = append(patterns, extra...)
	for _, p := range patterns {
		if !doublestar.ValidatePattern(p) {
			return nil, doublestar.ErrBadPattern
		}
	}
	return &Matcher{patterns: patterns}, nil
}

func (m *Matcher) Match(rel string) bool {
	rel = path.Clean("/" + strings.ReplaceAll(rel, "\\", "/"))
	rel = strings.TrimPrefix(rel, "/")
	if rel == "." || rel == "" {
		return false
	}
	parts := strings.Split(rel, "/")
	// Match full path and each path prefix (so ignored dirs are skipped while walking).
	var acc string
	for i, part := range parts {
		if i == 0 {
			acc = part
		} else {
			acc = acc + "/" + part
		}
		for _, pat := range m.patterns {
			ok, err := doublestar.Match(pat, acc)
			if err == nil && ok {
				return true
			}
		}
	}
	return false
}
```

```go
package catalog

import (
	"fmt"
	"path"
	"path/filepath"
	"strings"
)

func IsHTML(name string) bool {
	ext := strings.ToLower(filepath.Ext(name))
	return ext == ".html" || ext == ".htm"
}

func NormalizeRel(p string) (string, error) {
	p = strings.ReplaceAll(p, "\\", "/")
	p = path.Clean("/" + p)
	p = strings.TrimPrefix(p, "/")
	if p == ".." || strings.HasPrefix(p, "../") || strings.Contains(p, "/../") {
		return "", fmt.Errorf("invalid path %q", p)
	}
	if p == "." {
		return "", nil
	}
	return p, nil
}

func ResolveUnderRoot(root, rel string) (string, error) {
	rel, err := NormalizeRel(rel)
	if err != nil {
		return "", err
	}
	absRoot, err := filepath.Abs(root)
	if err != nil {
		return "", err
	}
	abs := filepath.Join(absRoot, filepath.FromSlash(rel))
	abs, err = filepath.Abs(abs)
	if err != nil {
		return "", err
	}
	sep := string(filepath.Separator)
	if abs != absRoot && !strings.HasPrefix(abs, absRoot+sep) {
		return "", fmt.Errorf("path escapes root")
	}
	return abs, nil
}
```

```go
package main

import (
	"context"
	"flag"
	"fmt"
	"net"
	"net/http"
	"os"
	"os/signal"
	"path/filepath"

	"mino/internal/catalog"
	"mino/internal/config"
	"mino/internal/ignore"
	"mino/internal/server"
	"mino/internal/watcher"
)

func main() {
	portFlag := flag.Int("port", -1, "override listen port")
	flag.Parse()
	root := "."
	if flag.NArg() > 0 {
		root = flag.Arg(0)
	}
	abs, err := filepath.Abs(root)
	if err != nil {
		fail(err)
	}
	info, err := os.Stat(abs)
	if err != nil || !info.IsDir() {
		fail(fmt.Errorf("root must be an existing directory: %s", abs))
	}
	cfg, err := config.Load(abs)
	if err != nil {
		fail(err)
	}
	if *portFlag >= 0 {
		cfg.Port = *portFlag
	}
	matcher, err := ignore.New(cfg.Ignore)
	if err != nil {
		fail(err)
	}
	cat := catalog.New(abs, matcher)
	if err := cat.Scan(); err != nil {
		fail(err)
	}
	hub := server.NewHub()
	srv := server.New(abs, cfg.Name, cat, hub)
	watchEnabled := true
	w, err := watcher.Start(abs, cat, func(evs []catalog.Event) {
		for _, e := range evs {
			hub.Publish(e)
		}
	})
	if err != nil {
		fmt.Fprintf(os.Stderr, "mino: watcher disabled: %v\n", err)
		watchEnabled = false
	} else {
		defer w.Close()
	}
	srv.SetWatchEnabled(watchEnabled)

	addr := fmt.Sprintf("%s:%d", cfg.Host, cfg.Port)
	ln, err := net.Listen("tcp", addr)
	if err != nil {
		fail(err)
	}
	url := fmt.Sprintf("http://%s/", ln.Addr().String())
	fmt.Println(url)

	httpSrv := &http.Server{Handler: srv.Handler()}
	go func() {
		if err := httpSrv.Serve(ln); err != nil && err != http.ErrServerClosed {
			fail(err)
		}
	}()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()
	<-ctx.Done()
	_ = httpSrv.Shutdown(context.Background())
}

func fail(err error) {
	fmt.Fprintf(os.Stderr, "mino: %v\n", err)
	os.Exit(1)
}
```
